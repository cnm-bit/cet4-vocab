<script>
const wordList = [
    {en:"abandon",cn:"抛弃，放弃"},
    {en:"absorb",cn:"吸收；使全神贯注"},
    {en:"access",cn:"通道；使用权"},

    {en:"assess",cn:"评估，评价"},

    {en:"authority",cn:"权威；当局"},
    {en:"available",cn:"可获得的；有空的"},

    {en:"campaign",cn:"运动；战役"},
    {en:"challenge",cn:"挑战"},
    {en:"circumstance",cn:"情况，环境"},
    {en:"concentrate",cn:"集中，专心"},
    {en:"consequence",cn:"后果，结果"},
    {en:"considerable",cn:"相当大的；重要的"},
    {en:"consume",cn:"消耗；消费"},
    {en:"contribute",cn:"贡献；促成"},
    {en:"convince",cn:"说服，使确信"},
    {en:"decline",cn:"下降；谢绝"},
    {en:"defect",cn:"缺点，缺陷"},

    {en:"diminish",cn:"减少，降低"}
];

let score = 0;
let currentIndex = 0;
let questions = [];
let answered = false;
let gameMode = "en2cn";
let useErrorBank = false;

const scoreDom = document.getElementById('score');
const currentDom = document.getElementById('current');
const enWordDom = document.getElementById('enWord');
const cnDescDom = document.getElementById('cnDesc');
const tipTextDom = document.getElementById('tipText');
const optionBox = document.getElementById('optionBox');
const spellBox = document.getElementById('spellBox');
const spellInput = document.getElementById('spellInput');
const spellMsg = document.getElementById('spellMsg');
const nextBtn = document.getElementById('nextBtn');
const gameView = document.getElementById('gameView');
const resultView = document.getElementById('resultView');
const finalScoreDom = document.getElementById('finalScore');
const noteBookView = document.getElementById('noteBookView');
const errorListWrap = document.getElementById("errorListWrap");
const modeBtns = document.querySelectorAll('.mode-btn');

function getErrorWords(){
    const raw = localStorage.getItem("cet4ErrorWords");
    return raw ? JSON.parse(raw) : [];
}
function saveErrorWord(wordObj){
    let arr = getErrorWords();
    const key = wordObj.en;
    const exist = arr.find(x=>x.en === key);
    if(!exist){
        arr.push(wordObj);
        localStorage.setItem("cet4ErrorWords",JSON.stringify(arr));
    }
}
function clearAllError(){
    localStorage.removeItem("cet4ErrorWords");
    renderNoteBook();
}
function renderNoteBook(){
    const list = getErrorWords();
    errorListWrap.innerHTML = "";
    if(list.length === 0){
        errorListWrap.innerHTML = `<div class="empty-tip">错题本为空，快去做题吧</div>`;
        return;
    }
    list.forEach(w=>{
        const div = document.createElement("div");
        div.className="error-item";
        div.innerHTML = `<div class="error-en">${w.en}</div><div class="error-cn">${w.cn}</div>`;
        errorListWrap.appendChild(div);
    })
}

document.getElementById("openNoteBook").onclick = function(){
    gameView.classList.add("hide");
    resultView.style.display="none";
    noteBookView.style.display="block";
    renderNoteBook();
}
document.getElementById("backGame").onclick = function(){
    noteBookView.style.display="none";
    gameView.classList.remove("hide");
}
document.getElementById("clearError").onclick = clearAllError;
document.getElementById("startErrorQuiz").onclick = function(){
    const errList = getErrorWords();
    if(errList.length===0){alert("错题本是空的！");return;}
    useErrorBank = true;
    noteBookView.style.display="none";
    gameView.classList.remove("hide");
    startGame();
}

modeBtns.forEach(btn=>{
    btn.onclick = ()=>{
        modeBtns.forEach(b=>b.classList.remove('active'));
        btn.classList.add('active');
        gameMode = btn.dataset.mode;
        useErrorBank = false;
        startGame();
    }
})

function shuffle(arr){
    let temp = [...arr];
    for(let i=temp.length-1;i>0;i--){
        const j = Math.floor(Math.random()*(i+1));
        [temp[i],temp[j]] = [temp[j],temp[i]];
    }
    return temp;
}

function buildOptions(rightWord,mode){
    if(mode === "en2cn"){
        const wrongPool = wordList.filter(w=>w.cn !== rightWord.cn);
        const wrongs = shuffle(wrongPool).slice(0,3);
        return shuffle([rightWord,...wrongs]);
    }else if(mode === "cn2en"){
        const wrongPool = wordList.filter(w=>w.en !== rightWord.en);
        const wrongs = shuffle(wrongPool).slice(0,3);
        return shuffle([rightWord,...wrongs]);
    }
}

function renderQuestion(){
    answered = false;
    nextBtn.disabled = true;
    spellMsg.innerText = "";
    spellInput.value = "";
    const q = questions[currentIndex];
    currentDom.innerText = currentIndex+1;
    optionBox.classList.remove("hide");
    spellBox.classList.add("hide");

    if(gameMode === "en2cn"){
        enWordDom.innerText = q.en;
        cnDescDom.innerText = "";
        tipTextDom.innerText = "看英文，选出中文释义";
        optionBox.innerHTML = "";
        const opts = buildOptions(q,"en2cn");
        opts.forEach(item=>{
            const btn = document.createElement('button');
            btn.className = "option-btn";
            btn.innerText = item.cn;
            btn.onclick = ()=>{
                if(answered) return;answered=true;nextBtn.disabled=false;
                if(item.cn === q.cn){
                    btn.classList.add('right');score++;scoreDom.innerText=score;
                }else{
                    btn.classList.add('wrong');
                    optionBox.querySelectorAll('.option-btn').forEach(b=>{
                        if(b.innerText === q.cn) b.classList.add('right');
                    })
                    saveErrorWord(q);
                }
            }
            optionBox.appendChild(btn);
        })
    }else if(gameMode === "cn2en"){
        enWordDom.innerText = "";
        cnDescDom.innerText = q.cn;
        tipTextDom.innerText = "看中文，选出对应的英文单词";
        optionBox.innerHTML = "";
        const opts = buildOptions(q,"cn2en");
        opts.forEach(item=>{
            const btn = document.createElement('button');
            btn.className = "option-btn";
            btn.innerText = item.en;
            btn.onclick = ()=>{
                if(answered) return;answered=true;nextBtn.disabled=false;
                if(item.en === q.en){
                    btn.classList.add('right');score++;scoreDom.innerText=score;
                }else{
                    btn.classList.add('wrong');
                    optionBox.querySelectorAll('.option-btn').forEach(b=>{
                        if(b.innerText === q.en) b.classList.add('right');
                    })
                    saveErrorWord(q);
                }
            }
            optionBox.appendChild(btn);
        })
    }else if(gameMode === "spell"){
        enWordDom.innerText = "????";
        cnDescDom.innerText = q.cn;
        tipTextDom.innerText = "根据释义拼写英文单词";
        optionBox.classList.add("hide");
        spellBox.classList.remove("hide");
        spellInput.focus();
        spellInput.onkeydown = (e)=>{
            if(e.key==='Enter' && !answered){
                checkSpell(q);
            }
        }
    }
}

function checkSpell(q){
    answered = true;
    nextBtn.disabled = false;
    const userAns = spellInput.value.trim().toLowerCase();
    const real = q.en.toLowerCase();
    if(userAns === real){
        spellMsg.innerText = "✅拼写正确";
        spellMsg.className = "spell-msg msg-right";
        score++;scoreDom.innerText=score;
    }else{
        spellMsg.innerText = `❌错误，正确单词：${q.en}`;
        spellMsg.className = "spell-msg msg-wrong";
        saveErrorWord(q);
    }
}

nextBtn.onclick = function(){
    if(gameMode==="spell" && !answered){
        checkSpell(questions[currentIndex]);
        return;
    }
    currentIndex++;
    if(currentIndex >= questions.length){
        showResult();
    }else{
        renderQuestion();
    }
}

function showResult(){
    gameView.classList.add("hide");
    resultView.style.display = 'block';
    finalScoreDom.innerText = score;
}

function startGame(){
    score = 0;
    currentIndex = 0;
    scoreDom.innerText = 0;
    gameView.classList.remove("hide");
    resultView.style.display = 'none';
    noteBookView.style.display="none";
    let sourceList;
    if(useErrorBank){
        sourceList = getErrorWords();
    }else{
        sourceList = wordList;
    }
    questions = shuffle(sourceList).slice(0,15);
    renderQuestion();
}
startGame();
